function formatKz(value) {
  return value.toLocaleString("pt-AO");
}

function findPlanByAmount(amount) {
  return PLANS.find((p) => amount >= p.min && amount <= p.max) || null;
}

// Calculates two payout modes: monthly payouts vs end-of-term (capital+profit reinvest disabled)
function calculateReturns(amount, plan, mode = "end") {
  if (!plan) return null;
  const monthlyRate = plan.monthlyPct / 100;
  const months = plan.months;

  if (mode === "monthly") {
    // monthly payouts: simple interest paid each month
    const monthlyProfit = amount * monthlyRate;
    const totalProfit = monthlyProfit * months;
    const total = amount + totalProfit;
    return {
      monthlyProfit,
      totalProfit,
      total,
    };
  } else {
    // end-of-term: compound each month -> capital grows each month
    let capital = amount;
    for (let i = 0; i < months; i++) {
      capital = capital * (1 + monthlyRate);
    }
    const total = Math.round(capital);
    const totalProfit = total - amount;
    return {
      monthlyProfit: null,
      totalProfit,
      total,
    };
  }
}
